⚙ Cấu hình nguồn dữ liệu
Hiện tại tool thử lấy giá theo thứ tự:
1. Direct fetch (không proxy)
2. CORS proxy (allorigins, corsproxy.io…)
3. Google Apps Script URL (nếu bạn có cấu hình)
📌 Google Apps Script URL (khuyên dùng)
Ổn định nhất, không bị CORS. Setup ~10 phút, miễn phí.
📋 Cách tạo Google Apps Script (bấm để xem)
Bước 1: Mở script.google.com → New project
Bước 2: Xoá code mặc định, paste code bên dưới
Bước 3: Deploy → New deployment → Web app → Execute as: Me → Who has access: Anyone → Deploy
Bước 4: Copy URL, paste vào ô trên

function doGet(e) { var shop = e.parameter.shop || 'sjc'; var result = {}; try { if (shop === 'sjc') { var xml = UrlFetchApp.fetch('https://sjc.com.vn/xml/tygiavang.xml').getContentText(); var doc = XmlService.parse(xml); var items = doc.getRootElement().getChildren(); for (var i = 0; i < items.length; i++) { var name = items[i].getAttribute('type') ? items[i].getAttribute('type').getValue() : items[i].getText(); if (/nh[aâ]n|999/i.test(name)) { result.buy = parseFloat(items[i].getAttribute('buy').getValue()); result.sell = parseFloat(items[i].getAttribute('sell') ? items[i].getAttribute('sell').getValue() : items[i].getAttribute('h') ? items[i].getAttribute('h').getValue() : 0); break; } } } } catch(err) { result.error = err.message; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
